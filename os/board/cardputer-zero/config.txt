# Catalyst Linux on the M5Stack CardputerZero (Raspberry Pi CM0).

kernel=Image
arm_64bit=1

# The firmware keeps 64 MB for itself, as M5Stack ships it: the camera's
# image processing runs in the firmware and needs it (with 16 MB and the
# cut-down firmware the camera does not appear). The GPU's 3D core is driven
# by Linux (the vc4 driver) from its own contiguous memory pool, 32 MB, which
# the rest of the system borrows while the GPU and camera do not need it. No
# HDMI: the panel is an SPI display; the GPU draws into memory and the frame
# goes out over SPI.
gpu_mem=64
dtoverlay=vc4-kms-v3d,nohdmi,noaudio,cma-size=33554432
disable_splash=1
disable_overscan=1
boot_delay=0
# The watchdog: a system that hangs while it starts is restarted, which
# after an update tried once means back to the one that worked.
dtparam=watchdog=on

dtparam=i2c_arm=on
dtparam=spi=on
# The camera: the IMX219, named rather than detected, as M5Stack sets it.
camera_auto_detect=0
dtoverlay=imx219

# Serial console on GPIO14/15 (the Grove port). This takes the PL011 from
# Bluetooth. Switched on in Settings > Bluetooth, the radio gets the mini
# UART instead (catalyst-bt, added below the board's lines with the core
# clock held for it), and the console stays here, on ttyAMA0.
enable_uart=1
dtoverlay=disable-bt

# USB device mode, for a link to the Systemcore or a PC over USB-C.
dtoverlay=dwc2

# The speaker's amplifier (enabled by GPIO24) held off from the first
# instant: idle it hisses and draws about 9 mA. Once the sound drivers load
# (catalyst-late), the kernel switches it on only while something plays,
# and off again five seconds after the last sound.
gpio=24=op,dl

# The board: I/O expander, panel, backlight, keyboard, fuel gauge.
dtoverlay=cardputerzero-v5-overlay
dtoverlay=bq27220_v5

# Infrared, as M5Stack sets it up: the receiver on GPIO13 (pulled up; its
# output is low while it sees light) and the LED on GPIO12, driven by the
# kernel's software transmitter. Neither driver loads at start: the Remote
# app loads the transmitter while it is open and the receiver only while a
# button is learned (catalyst.conf in modprobe.d). The receiver gets the
# empty key map, so no remote in the room is ever taken for a keyboard.
dtoverlay=gpio-ir,gpio_pin=13,gpio_pull=up,rc-map-name=rc-empty
dtoverlay=gpio-ir-tx,gpio_pin=12
