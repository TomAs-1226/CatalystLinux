# Catalyst Linux on the M5Stack CardputerZero (Raspberry Pi CM0).

start_file=start_cd.elf
fixup_file=fixup_cd.dat
kernel=Image
arm_64bit=1

# The firmware keeps 16 MB for itself. The GPU's 3D core is driven by Linux
# (the vc4 driver) from its own contiguous memory pool, 32 MB, which the rest
# of the system borrows while the GPU does not need it. No HDMI: the panel is
# an SPI display; the GPU draws into memory and the frame goes out over SPI.
gpu_mem=16
dtoverlay=vc4-kms-v3d,nohdmi,noaudio,cma-size=33554432
disable_splash=1
disable_overscan=1
boot_delay=0
# The watchdog: a system that hangs while it starts is restarted, which
# after an update tried once means back to the one that worked.
dtparam=watchdog=on

dtparam=i2c_arm=on
dtparam=spi=on
camera_auto_detect=0

# Serial console on GPIO14/15 (the Grove port). This takes the PL011 from
# Bluetooth, which nothing uses yet.
enable_uart=1
dtoverlay=disable-bt

# USB device mode, for a link to the Systemcore or a PC over USB-C.
dtoverlay=dwc2

# The board: I/O expander, panel, backlight, keyboard, fuel gauge.
dtoverlay=cardputerzero-v5-overlay
dtoverlay=bq27220_v5
